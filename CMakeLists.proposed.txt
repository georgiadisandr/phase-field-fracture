# -----------------------------------------------------------------------------
# [NEW] Optimization
# -----------------------------------------------------------------------------
# EIGEN_NO_DEBUG strips Eigen's internal size/alignment assertions. Release
# already defines NDEBUG, which does most of this, but being explicit is free.
target_compile_definitions(phasefield_1 PRIVATE $<$<CONFIG:Release>:EIGEN_NO_DEBUG>)

if(MSVC)
    target_compile_options(phasefield_1 PRIVATE
        $<$<CONFIG:Release>:/O2 /arch:AVX2 /fp:contract>)
else()
    # -march=native is the important one: without it GCC targets baseline
    # x86-64 (SSE2 only) and Eigen never uses AVX2/FMA. Swap for
    # -march=x86-64-v3 if the binary must run on another machine.
    #
    # DELIBERATELY NOT USING -ffast-math: it implies -ffinite-math-only, which
    # lets the compiler assume NaN/Inf cannot occur. Solver.cpp relies on
    # delta.allFinite() to catch a diverging Newton increment, and that check
    # can be optimized away. -ffp-contract=fast gives you FMA, which is where
    # most of the -ffast-math benefit is for this code, without that hazard.
    target_compile_options(phasefield_1 PRIVATE
        $<$<CONFIG:Release>:-O3 -march=native -funroll-loops -ffp-contract=fast>)
endif()

# Link-time optimization. Checked rather than assumed, since MinGW setups vary.
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo_ok OUTPUT ipo_msg)
if(ipo_ok)
    set_property(TARGET phasefield_1
                 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    message(STATUS "LTO/IPO: enabled for Release")
else()
    message(STATUS "LTO/IPO: not available (${ipo_msg})")
endif()

# -----------------------------------------------------------------------------
# [NEW] OpenMP -- optional, off by default. Nothing in the code is threaded yet,
# so this only matters once the element assembly loop is parallelized.
# -----------------------------------------------------------------------------
option(PFM_USE_OPENMP "Enable OpenMP (for threaded element assembly)" OFF)
if(PFM_USE_OPENMP)
    find_package(OpenMP)
    if(OpenMP_CXX_FOUND)
        target_link_libraries(phasefield_1 PRIVATE OpenMP::OpenMP_CXX)
        target_compile_definitions(phasefield_1 PRIVATE PFM_OPENMP)
        message(STATUS "OpenMP: enabled")
    else()
        message(WARNING "PFM_USE_OPENMP=ON but OpenMP was not found")
    endif()
endif()

# -----------------------------------------------------------------------------
# Windows: copy the gmsh DLL next to the executable
# -----------------------------------------------------------------------------
if(WIN32)
    find_file(GMSH_DLL NAMES gmsh-4.15.dll gmsh.dll
        HINTS "${GMSH_LIB_DIR}"
        PATH_SUFFIXES bin lib)
    if(GMSH_DLL)
        add_custom_command(TARGET phasefield_1 POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "${GMSH_DLL}" "$<TARGET_FILE_DIR:phasefield_1>")
    else()
        message(WARNING "gmsh DLL not found; phasefield_1.exe will fail to start at runtime")
    endif()
endif()
